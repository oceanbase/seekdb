cmake_minimum_required(VERSION 3.18)
project(seekdb_graph_p0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_executable(graph_p0 main.cpp)
add_executable(graph_p0_unit test.cpp)
foreach(target graph_p0 graph_p0_unit)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Werror)
endforeach()
option(GRAPH_P0_MYSQL "Build the real seekdb access driver" ON)
if(GRAPH_P0_MYSQL)
  find_path(MYSQL_INCLUDE_DIR mysql.h PATH_SUFFIXES mysql mariadb REQUIRED)
  find_library(MYSQL_LIBRARY NAMES mariadb mysqlclient libmariadb.so.3 REQUIRED)
  target_include_directories(graph_p0 PRIVATE ${MYSQL_INCLUDE_DIR})
  target_link_libraries(graph_p0 PRIVATE ${MYSQL_LIBRARY})
  target_compile_definitions(graph_p0 PRIVATE GRAPH_P0_MYSQL)
endif()
enable_testing()
add_test(NAME graph_p0_unit COMMAND graph_p0_unit)
